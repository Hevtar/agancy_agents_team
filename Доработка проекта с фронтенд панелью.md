# 4. 🌐 Nginx конфигурация

```nginx
# deploy/nginx/nginx.conf

user nginx;
worker_processes auto;
error_log /var/log/nginx/error.log warn;
pid /var/run/nginx.pid;

events {
    worker_connections 1024;
    use epoll;
    multi_accept on;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    # ── Логирование ─────────────────────────────────────────
    log_format main '$remote_addr - $remote_user [$time_local] '
                    '"$request" $status $body_bytes_sent '
                    '"$http_referer" "$http_user_agent"';
    access_log /var/log/nginx/access.log main;

    # ── Оптимизация ─────────────────────────────────────────
    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;
    client_max_body_size 50M;
    gzip on;
    gzip_types text/plain text/css application/json
               application/javascript text/xml
               application/xml application/xml+rss text/javascript;

    # ── Rate limiting ────────────────────────────────────────
    limit_req_zone $binary_remote_addr
        zone=api_limit:10m rate=30r/m;
    limit_req_zone $binary_remote_addr
        zone=login_limit:10m rate=5r/m;

    # ── Upstream серверы ─────────────────────────────────────
    upstream api_backend {
        server api:8000;
        keepalive 32;
    }

    upstream frontend_backend {
        server frontend:3001;
        keepalive 32;
    }

    upstream grafana_backend {
        server grafana:3000;
    }

    # ════════════════════════════════════════════════════════
    # HTTP → HTTPS redirect
    # ════════════════════════════════════════════════════════
    server {
        listen 80;
        server_name ${DOMAIN} www.${DOMAIN};

        # Let's Encrypt challenge
        location /.well-known/acme-challenge/ {
            root /var/www/certbot;
        }

        # Всё остальное → HTTPS
        location / {
            return 301 https://$host$request_uri;
        }
    }

    # ════════════════════════════════════════════════════════
    # HTTPS основной сервер
    # ════════════════════════════════════════════════════════
    server {
        listen 443 ssl http2;
        server_name ${DOMAIN} www.${DOMAIN};

        # ── SSL ─────────────────────────────────────────────
        ssl_certificate /etc/letsencrypt/live/${DOMAIN}/fullchain.pem;
        ssl_certificate_key /etc/letsencrypt/live/${DOMAIN}/privkey.pem;
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers ECDHE-RSA-AES256-GCM-SHA512:DHE-RSA-AES256-GCM-SHA512
                    :ECDHE-RSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-SHA384;
        ssl_prefer_server_ciphers on;
        ssl_session_cache shared:SSL:10m;
        ssl_session_timeout 10m;
        ssl_stapling on;
        ssl_stapling_verify on;

        # ── Security headers ─────────────────────────────────
        add_header Strict-Transport-Security
            "max-age=31536000; includeSubDomains" always;
        add_header X-Frame-Options DENY always;
        add_header X-Content-Type-Options nosniff always;
        add_header X-XSS-Protection "1; mode=block" always;
        add_header Referrer-Policy "strict-origin-when-cross-origin";

        # ── Frontend (Next.js) ───────────────────────────────
        location / {
            proxy_pass http://frontend_backend;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection 'upgrade';
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_cache_bypass $http_upgrade;
            proxy_read_timeout 60s;
        }

        # ── REST API ─────────────────────────────────────────
        location /api/v1/ {
            limit_req zone=api_limit burst=20 nodelay;

            proxy_pass http://api_backend;
            proxy_http_version 1.1;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_read_timeout 300s;
            proxy_send_timeout 300s;
        }

        # ── Login endpoint — строгий rate limit ──────────────
        location /api/v1/auth/token {
            limit_req zone=login_limit burst=3 nodelay;
            proxy_pass http://api_backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        # ── WebSocket (для real-time обновлений) ─────────────
        location /ws/ {
            proxy_pass http://api_backend;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
            proxy_set_header Host $host;
            proxy_read_timeout 3600s;
        }

        # ── Grafana ──────────────────────────────────────────
        location /grafana/ {
            proxy_pass http://grafana_backend/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
        }

        # ── Prometheus (только с локальной сети) ─────────────
        location /prometheus/ {
            allow 127.0.0.1;
            allow 172.20.0.0/16;
            deny all;
            proxy_pass http://prometheus:9090/;
        }

        # ── Healthcheck (публичный) ───────────────────────────
        location /health {
            proxy_pass http://api_backend/health;
            access_log off;
        }
    }
}
```

```typescript
// frontend/src/app/api/auth/[...nextauth]/route.ts

import NextAuth, { NextAuthOptions } from 'next-auth'
import CredentialsProvider from 'next-auth/providers/credentials'
import axios from 'axios'

const AGENCY_API_URL = process.env.AGENCY_API_URL || 'http://api:8000/api/v1'

export const authOptions: NextAuthOptions = {
  providers: [
    CredentialsProvider({
      name: 'Agency AI',
      credentials: {
        username: {
          label: 'Логин',
          type: 'text',
          placeholder: 'admin'
        },
        password: {
          label: 'Пароль',
          type: 'password'
        },
      },
      async authorize(credentials) {
        if (!credentials?.username || !credentials?.password) {
          throw new Error('Введите логин и пароль')
        }

        try {
          // Получаем JWT токен от Agency API
          const response = await axios.post(
            `${AGENCY_API_URL}/auth/token`,
            new URLSearchParams({
              username: credentials.username,
              password: credentials.password,
            }),
            {
              headers: {
                'Content-Type': 'application/x-www-form-urlencoded',
              },
            }
          )

          const { access_token, token_type, user } = response.data

          if (access_token) {
            return {
              id: user.id,
              name: user.username,
              email: user.email,
              role: user.role,
              accessToken: access_token,
            }
          }
          return null
        } catch (error: any) {
          const message =
            error.response?.data?.detail || 'Неверный логин или пароль'
          throw new Error(message)
        }
      },
    }),
  ],

  session: {
    strategy: 'jwt',
    maxAge: 24 * 60 * 60, // 24 часа
  },

  pages: {
    signIn: '/login',
    error: '/login',
  },

  callbacks: {
    async jwt({ token, user }) {
      if (user) {
        token.accessToken = (user as any).accessToken
        token.role = (user as any).role
      }
      return token
    },
    async session({ session, token }) {
      session.accessToken = token.accessToken as string
      session.user.role = token.role as string
      return session
    },
  },

  secret: process.env.NEXTAUTH_SECRET,
}

const handler = NextAuth(authOptions)
export { handler as GET, handler as POST }
```

```typescript
                  className="bg-gray-900 border border-gray-800
```

## 6.3 Bot Middleware (аутентификация)

```python
# bot/middlewares/auth.py

from typing import Any, Awaitable, Callable, Dict
from aiogram import BaseMiddleware
from aiogram.types import TelegramObject, Message, CallbackQuery
import structlog

from bot.config import config

logger = structlog.get_logger()


class AuthMiddleware(BaseMiddleware):
    """
    Проверяет, что пользователь входит в список разрешённых.
    Если список пуст — доступ открыт всем (небезопасно для прода).
    """

    async def __call__(
        self,
        handler: Callable[[TelegramObject, Dict[str, Any]], Awaitable[Any]],
        event: TelegramObject,
        data: Dict[str, Any],
    ) -> Any:
        # Получаем user из события
        user = None
        if isinstance(event, Message):
            user = event.from_user
        elif isinstance(event, CallbackQuery):
            user = event.from_user

        if user is None:
            return await handler(event, data)

        allowed = config.allowed_user_ids

        # Если список не задан — пропускаем всех
        if not allowed:
            data["is_admin"] = True
            return await handler(event, data)

        # Проверяем доступ
        if user.id not in allowed:
            logger.warning(
                "unauthorized_access",
                user_id=user.id,
                username=user.username,
            )
            if isinstance(event, Message):
                await event.answer(
                    "⛔ <b>Доступ запрещён</b>\n\n"
                    "Вы не авторизованы для использования этого бота.\n"
                    f"Ваш Telegram ID: <code>{user.id}</code>\n\n"
                    "Обратитесь к администратору системы.",
                    parse_mode="HTML",
                )
            elif isinstance(event, CallbackQuery):
                await event.answer(
                    "⛔ Доступ запрещён", show_alert=True
                )
            return  # Прерываем обработку

        # Устанавливаем флаг админа
        data["is_admin"] = user.id in config.admin_user_ids
        return await handler(event, data)
```

```python
import aiohttp
from typing import Any, Dict, List, Optional
from tenacity import (
    retry,
    stop_after_attempt,
    wait_exponential,
    retry_if_exception_type,
)
import structlog

from bot.config import config

logger = structlog.get_logger()


class AgencyAPIService:
    """HTTP клиент для обращения к Agency FastAPI."""

    def __init__(self):
        self.base_url = config.agency_api_url.rstrip("/")
        self.timeout = aiohttp.ClientTimeout(
            total=config.request_timeout
        )
        self._session: Optional[aiohttp.ClientSession] = None

    async def get_session(self) -> aiohttp.ClientSession:
        if self._session is None or self._session.closed:
            headers = {}
            if config.agency_api_key:
                headers["X-API-Key"] = config.agency_api_key
            self._session = aiohttp.ClientSession(
                headers=headers,
                timeout=self.timeout,
            )
        return self._session

    async def close(self):
        if self._session and not self._session.closed:
            await self._session.close()

    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=1, max=5),
        retry=retry_if_exception_type(aiohttp.ClientError),
        reraise=True,
    )
    async def _get(
        self,
        path: str,
        params: Optional[Dict] = None,
    ) -> Any:
        session = await self.get_session()
        url = f"{self.base_url}{path}"
        async with session.get(url, params=params) as resp:
            resp.raise_for_status()
            return await resp.json()

    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=1, max=5),
        retry=retry_if_exception_type(aiohttp.ClientError),
        reraise=True,
    )
    async def _post(
        self,
        path: str,
        data: Optional[Dict] = None,
    ) -> Any:
        session = await self.get_session()
        url = f"{self.base_url}{path}"
        async with session.post(url, json=data or {}) as resp:
            resp.raise_for_status()
            return await resp.json()
```

```python
from typing import Dict, Optional
from datetime import datetime


def fmt_health(health: str) -> str:
    return {
        "green": "🟢 Норма",
        "yellow": "🟡 Внимание",
        "red": "🔴 Критично",

def fmt_severity(severity: str) -> str:
    return {
        "CRITICAL": "🔴 CRITICAL",
        "HIGH": "🟠 HIGH",
        "MEDIUM": "🟡 MEDIUM",
        "LOW": "🟢 LOW",
    }.get(severity, severity)


def fmt_task_status(status: str) -> str:
    return {
        "pending": "⏳ Ожидает",
        "in_progress": "⚡ В работе",
        "done": "✅ Выполнено",
        "failed": "❌ Ошибка",
        "cancelled": "🚫 Отменено",
    }.get(status, status)


def fmt_agent_status(status: str) -> str:
    return {
        "idle": "💤 Свободен",
        "busy": "⚡ Занят",
        "error": "❌ Ошибка",
        "paused": "⏸ Пауза",
    }.get(status, status)


def fmt_date(dt_str: Optional[str]) -> str:
    if not dt_str:
        return "—"
    try:
        dt = datetime.fromisoformat(
            dt_str.replace("Z", "+00:00")
        )
        return dt.strftime("%d.%m.%Y %H:%M")
    except Exception:
        return dt_str


def fmt_tokens(n: int) -> str:
    if n >= 1_000_000:
        return f"{n / 1_000_000:.1f}M"
    if n >= 1_000:
        return f"{n / 1_000:.1f}K"
    return str(n)


def fmt_project(p: Dict) -> str:
    pct = round(
        (p.get("tokens_used", 0) / max(p.get("token_budget", 1), 1)) * 100
    )
    bar_filled = int(pct / 10)
    bar = "█" * bar_filled + "░" * (10 - bar_filled)

    deadline = ""
    if p.get("deadline"):
        deadline = (
            f"\n📅 Дедлайн: {fmt_date(p['deadline'])}"
        )

    return (
        f"📁 <b>{p['title']}</b>\n"
        f"🆔 <code>{p['project_id']}</code>\n"
        f"📊 Фаза: <b>{p.get('phase', '—')}</b>\n"
        f"❤️ Health: <b>{fmt_health(p.get('health', '—'))}</b>\n"
        f"⚡ Приоритет: {p.get('priority', '—')}\n"
        f"🪙 Токены: {fmt_tokens(p.get('tokens_used', 0))} / "
        f"{fmt_tokens(p.get('token_budget', 0))}\n"
        f"[{bar}] {pct}%"
        f"{deadline}"
    )


def fmt_task(t: Dict) -> str:
    error_part = ""
    if t.get("error_message"):
        error_part = (
            f"\n⚠️ Ошибка: <code>{t['error_message'][:100]}</code>"
        )

    return (
        f"📋 <b>{t['title']}</b>\n"
        f"🆔 <code>{t['task_id']}</code>\n"
        f"📌 Статус: <b>{fmt_task_status(t.get('status', '—'))}</b>\n"
        f"📁 Проект: <code>{t.get('project_id', '—')}</code>\n"
        f"🤖 Агент: <code>{t.get('assignee_id', '—')}</code>\n"
        f"⚡ Приоритет: {t.get('priority', '—')}\n"
        f"🪙 Токены: {fmt_tokens(t.get('tokens_used', 0))}\n"
        f"📅 Создана: {fmt_date(t.get('created_at'))}"
        f"{error_part}"
    )


def fmt_blocker(b: Dict) -> str:
    resolved_part = ""
    if b.get("resolved_at"):
        resolved_part = (
            f"\n✅ Решён: {fmt_date(b['resolved_at'])}"
        )

    return (
        f"🚧 <b>{b['title']}</b>\n"
        f"🆔 <code>{b['blocker_id']}</code>\n"
        f"🔥 Severity: <b>{fmt_severity(b.get('severity', '—'))}</b>\n"
        f"📌 Статус: <b>{b.get('status', '—')}</b>\n"
        f"📁 Проект: <code>{b.get('project_id', '—')}</code>\n"
        f"🤖 Обнаружил: <code>{b.get('detected_by', '—')}</code>\n"
        f"📝 {b.get('description', '')[:200]}\n"
        f"📅 Создан: {fmt_date(b.get('created_at'))}"
        f"{resolved_part}"
    )


def fmt_agent(a: Dict) -> str:
    success_rate = 0
    total = a.get("tasks_completed", 0) + a.get("tasks_failed", 0)
    if total > 0:
        success_rate = round(
            a["tasks_completed"] / total * 100
        )

    active = ""
    if a.get("active_task"):
        active = f"\n⚡ Задача: <code>{a['active_task'][:50]}</code>"

    return (
        f"🤖 <b>{a['name']}</b>\n"
        f"🆔 <code>{a['agent_id']}</code>\n"
        f"👥 Группа: {a.get('group', '—')}\n"
        f"🧠 Модель: {a.get('model', '—')}\n"
        f"📌 Статус: <b>{fmt_agent_status(a.get('status', '—'))}</b>\n"
        f"✅ Выполнено: {a.get('tasks_completed', 0)}\n"
        f"❌ Провалено: {a.get('tasks_failed', 0)}\n"
        f"📈 Успешность: {success_rate}%\n"
        f"🪙 Токены: {fmt_tokens(a.get('tokens_used', 0))}"
        f"{active}"
    )


def fmt_dashboard(s: Dict) -> str:
    token_pct = 0
    if s.get("daily_token_budget", 0) > 0:
        token_pct = round(
            s["tokens_used_today"] / s["daily_token_budget"] * 100
        )

    bar_filled = int(token_pct / 10)
    bar = "█" * bar_filled + "░" * (10 - bar_filled)

        "\n⚠️ <b>Есть RED проекты!</b>"

        else ""
    )
    crit_alert = (
        "\n🚨 <b>Есть CRITICAL блокеры!</b>"
        if s.get("critical_blockers", 0) > 0
```

```python
# bot/main.py

import asyncio
import structlog
from aiogram import Bot, Dispatcher
from aiogram.client.default import DefaultBotProperties
from aiogram.enums import ParseMode
from aiogram.fsm.storage.redis import RedisStorage
from aiogram.fsm.storage.memory import MemoryStorage

from bot.config import config
from bot.middlewares.auth import AuthMiddleware
from bot.services.api_service import api_service

# Handlers
from bot.handlers import (
    start,
    projects,
    tasks,
    agents,
    blockers,
    analytics,
)

logger = structlog.get_logger()


def setup_logging():
    structlog.configure(
        processors=[
            structlog.stdlib.add_log_level,
            structlog.stdlib.add_logger_name,
            structlog.processors.TimeStamper(fmt="iso"),
            structlog.dev.ConsoleRenderer(),
        ],
        wrapper_class=structlog.stdlib.BoundLogger,
        cache_logger_on_first_use=True,
    )


async def on_startup(bot: Bot):
    me = await bot.get_me()
    logger.info(
        "bot_started",
        username=me.username,
        id=me.id,
    )

    # Уведомляем админов о старте
    for admin_id in config.admin_user_ids:
        try:
            await bot.send_message(
                admin_id,
                "🟢 <b>Agency AI Bot запущен</b>\n\n"
                f"🤖 @{me.username}\n"
                f"🌐 API: {config.agency_api_url}",
                parse_mode="HTML",
            )
        except Exception as e:
            logger.warning(
                "admin_notify_failed",
                admin_id=admin_id,
                error=str(e),
            )


async def on_shutdown(bot: Bot):
    logger.info("bot_stopping")

    # Уведомляем админов об остановке
    for admin_id in config.admin_user_ids:
        try:
            await bot.send_message(
                admin_id,
                "🔴 <b>Agency AI Bot остановлен</b>",
                parse_mode="HTML",
            )
        except Exception:
            pass

    await api_service.close()
    logger.info("bot_stopped")


async def main():
    setup_logging()
    logger.info("bot_initializing")

    # Storage: Redis если доступен, иначе Memory
    try:
        storage = RedisStorage.from_url(config.redis_url)
        logger.info("storage_redis", url=config.redis_url)
    except Exception as e:
        logger.warning(
            "redis_unavailable",
            error=str(e),
            fallback="MemoryStorage",
        )
        storage = MemoryStorage()

    # Бот
    bot = Bot(
        token=config.telegram_bot_token,
        default=DefaultBotProperties(
            parse_mode=ParseMode.HTML,
        ),
    )

    # Диспетчер
    dp = Dispatcher(storage=storage)

    # Middleware
    dp.message.middleware(AuthMiddleware())
    dp.callback_query.middleware(AuthMiddleware())

    # Роутеры
    dp.include_router(start.router)
    dp.include_router(projects.router)
    dp.include_router(tasks.router)
    dp.include_router(agents.router)
    dp.include_router(blockers.router)
    dp.include_router(analytics.router)

    # Lifecycle hooks
    dp.startup.register(on_startup)
    dp.shutdown.register(on_shutdown)

    logger.info("bot_polling_start")

    try:
        await dp.start_polling(
            bot,
            allowed_updates=dp.resolve_used_update_types(),
            drop_pending_updates=True,
        )
    finally:
        await bot.session.close()


if __name__ == "__main__":
    asyncio.run(main())
```

```bash
#!/bin/bash
# ================================================================
# Agency AI — Deploy Script
# Использование: ./deploy.sh [--pull] [--migrate] [--restart]
# ================================================================

set -euo pipefail

# ── Цвета ────────────────────────────────────────────────────────
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m'

# ── Конфиг ───────────────────────────────────────────────────────
PROJECT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
COMPOSE_FILE="$PROJECT_DIR/docker-compose.yml"
ENV_FILE="$PROJECT_DIR/.env"
BACKUP_DIR="$PROJECT_DIR/backups"
TIMESTAMP=$(date +%Y%m%d_%H%M%S)

# ── Флаги ────────────────────────────────────────────────────────
DO_PULL=false
DO_MIGRATE=false
DO_RESTART=false

for arg in "$@"; do
    case $arg in
        --pull)     DO_PULL=true ;;
        --migrate)  DO_MIGRATE=true ;;
        --restart)  DO_RESTART=true ;;
        --all)
            DO_PULL=true
            DO_MIGRATE=true
            DO_RESTART=true
            ;;
    esac
done

# ── Утилиты ──────────────────────────────────────────────────────
log_info()    { echo -e "${BLUE}[INFO]${NC}  $1"; }
log_ok()      { echo -e "${GREEN}[OK]${NC}    $1"; }
log_warn()    { echo -e "${YELLOW}[WARN]${NC}  $1"; }
log_error()   { echo -e "${RED}[ERROR]${NC} $1"; }

check_requirements() {
    log_info "Проверяю зависимости..."
    for cmd in docker docker-compose curl; do
        if ! command -v "$cmd" &>/dev/null; then
            log_error "Не найден: $cmd"
            exit 1
        fi
    done
    log_ok "Все зависимости установлены"
}

check_env() {
    log_info "Проверяю .env файл..."
    if [[ ! -f "$ENV_FILE" ]]; then
        log_error ".env файл не найден!"
        log_warn "Скопируй: cp .env.example .env"
        exit 1
    fi

    # Проверяем критические переменные
    local required=(
        "POSTGRES_PASSWORD"
        "SECRET_KEY"
        "ANTHROPIC_API_KEY"
        "TELEGRAM_BOT_TOKEN"
        "NEXTAUTH_SECRET"
    )

    for var in "${required[@]}"; do
        val=$(grep "^${var}=" "$ENV_FILE" | cut -d= -f2-)
        if [[ -z "$val" || "$val" == *"CHANGE_ME"* ]]; then
            log_error "Не задана переменная: $var"
            exit 1
        fi
    done

    log_ok ".env файл в порядке"
}

backup_database() {
    log_info "Создаю резервную копию БД..."
    mkdir -p "$BACKUP_DIR"

    local backup_file="$BACKUP_DIR/db_$TIMESTAMP.sql.gz"

    if docker-compose -f "$COMPOSE_FILE" ps postgres \
        | grep -q "Up"; then
        docker-compose -f "$COMPOSE_FILE" exec -T postgres \
            pg_dump -U "$POSTGRES_USER" "$POSTGRES_DB" \
            | gzip > "$backup_file"
        log_ok "Бэкап: $backup_file"
    else
        log_warn "PostgreSQL не запущен — бэкап пропущен"
    fi

    # Удаляем старые бэкапы (старше 7 дней)
    find "$BACKUP_DIR" -name "db_*.sql.gz" -mtime +7 -delete
}

pull_images() {
    log_info "Скачиваю новые образы..."
    docker-compose -f "$COMPOSE_FILE" pull
    log_ok "Образы обновлены"
}

build_images() {
    log_info "Собираю образы..."
    docker-compose -f "$COMPOSE_FILE" build \
        --no-cache \
        --parallel
    log_ok "Образы собраны"
}

run_migrations() {
    log_info "Запускаю миграции БД..."

    # Убеждаемся что postgres запущен
    docker-compose -f "$COMPOSE_FILE" up -d postgres redis
    sleep 3

    docker-compose -f "$COMPOSE_FILE" run --rm api \
        alembic upgrade head

    log_ok "Миграции применены"
}

deploy_services() {
    log_info "Деплою сервисы..."

    # Graceful restart: поочерёдно
    local services=("api" "frontend" "bot" "celery_worker")

    for service in "${services[@]}"; do
        log_info "  Перезапускаю: $service"
        docker-compose -f "$COMPOSE_FILE" up -d \
            --no-deps \
            --force-recreate \
            "$service"
        sleep 2
    done

    log_ok "Все сервисы запущены"
}

health_check() {
    log_info "Проверяю здоровье сервисов..."
    local retries=10
    local wait=3

    for i in $(seq 1 $retries); do
        if curl -sf "http://localhost:8000/health" \
            -o /dev/null 2>&1; then
            log_ok "API отвечает ✓"
            break
        fi
        if [[ $i -eq $retries ]]; then
            log_error "API не отвечает после $retries попыток!"
            log_warn "Логи API:"
            docker-compose -f "$COMPOSE_FILE" logs \
                --tail=20 api
            exit 1
        fi
        log_warn "  Попытка $i/$retries — жду ${wait}с..."
        sleep $wait
    done

    # Проверяем frontend
    if curl -sf "http://localhost:3000" -o /dev/null 2>&1; then
        log_ok "Frontend отвечает ✓"
    else
        log_warn "Frontend не отвечает (может быть ещё стартует)"
    fi
}

show_status() {
    echo ""
    echo -e "${BLUE}════════════════════════════════════════${NC}"
    echo -e "${GREEN}  Agency AI — статус сервисов${NC}"
    echo -e "${BLUE}════════════════════════════════════════${NC}"
    docker-compose -f "$COMPOSE_FILE" ps
    echo ""
    log_ok "Деплой завершён в $(date '+%H:%M:%S')"
    echo ""
    echo -e "  🌐 Frontend:  ${GREEN}http://localhost:3000${NC}"
    echo -e "  🔌 API:       ${GREEN}http://localhost:8000${NC}"
    echo -e "  📚 API Docs:  ${GREEN}http://localhost:8000/docs${NC}"
    echo ""
}

# ── Источник .env для скриптов ────────────────────────────────────
load_env() {
    if [[ -f "$ENV_FILE" ]]; then
        export $(grep -v '^#' "$ENV_FILE" \
            | grep -v '^$' \
            | xargs 2>/dev/null) || true
    fi
}

# ── Main ─────────────────────────────────────────────────────────
main() {
    echo ""
    echo -e "${BLUE}╔══════════════════════════════════════╗${NC}"
    echo -e "${BLUE}║     Agency AI — Deploy Script        ║${NC}"
    echo -e "${BLUE}║     $(date '+%Y-%m-%d %H:%M:%S')           ║${NC}"
    echo -e "${BLUE}╚══════════════════════════════════════╝${NC}"
    echo ""

    cd "$PROJECT_DIR"

    check_requirements
    check_env
    load_env

    # Бэкап перед деплоем
    backup_database

    # Pull новых образов (если флаг)
    if [[ "$DO_PULL" == true ]]; then
        pull_images
    fi

    # Сборка
    build_images

    # Миграции (если флаг)
    if [[ "$DO_MIGRATE" == true ]]; then
        run_migrations
    fi

    # Деплой
    if [[ "$DO_RESTART" == true ]]; then
        deploy_services
    else
        docker-compose -f "$COMPOSE_FILE" up -d
    fi

    # Health check
    sleep 5
    health_check

    # Статус
    show_status
}

main "$@"
```

```makefile
		       {printf "  $(GREEN)%-18s$(NC) %s\n", $$1, $$2}'

	@pip install -r bot/requirements.txt

	@$(DC) -f docker-compose.yml up \

	@echo "$(GREEN)БД запущена. Запусти API и фронт вручную:$(NC)"
	@echo "  API:      uvicorn src.agency.main:app --reload"

	@cd frontend && npm run dev

dev-bot: ## Запустить только Bot
	@python -m bot.main

build: ## Собрать Docker образы
	@echo "$(BLUE)Сборка образов...$(NC)"
	@$(DC) build --parallel
	@echo "$(GREEN)Готово$(NC)"

up: ## Запустить все сервисы
	@$(DC) up -d
	@echo "$(GREEN)Сервисы запущены$(NC)"
	@echo "  Frontend: http://localhost:3000"
	@echo "  API:      http://localhost:8000"
	@echo "  Docs:     http://localhost:8000/docs"

down: ## Остановить все сервисы
	@$(DC) down

restart: ## Перезапустить все сервисы
	@$(DC) restart

restart-api: ## Перезапустить только API
	@$(DC) restart api

restart-bot: ## Перезапустить только Bot
	@$(DC) restart bot

logs: ## Показать логи всех сервисов
	@$(DC) logs -f --tail=100

logs-api: ## Логи API
	@$(DC) logs -f --tail=100 api

logs-bot: ## Логи Bot
	@$(DC) logs -f --tail=100 bot

logs-frontend: ## Логи Frontend
	@$(DC) logs -f --tail=100 frontend

	@$(DC) ps

# ── База данных ──────────────────────────────────────────────────
migrate: ## Применить миграции
	@echo "$(BLUE)Применяю миграции...$(NC)"
	@$(DC) exec api alembic upgrade head
	@echo "$(GREEN)Готово$(NC)"

migrate-create: ## Создать новую миграцию (make migrate-create MSG="...")
	@$(DC) exec api alembic revision \
		--autogenerate \
		-m "$(MSG)"

migrate-down: ## Откатить последнюю миграцию
	@$(DC) exec api alembic downgrade -1

db-shell: ## Открыть psql
	@$(DC) exec postgres psql \
		-U $${POSTGRES_USER:-agency_user} \
		-d $${POSTGRES_DB:-agency_db}

	@mkdir -p backups
	@$(DC) exec -T postgres pg_dump \
		-U $${POSTGRES_USER:-agency_user} \
		$${POSTGRES_DB:-agency_db} \
		| gzip > backups/db_$$(date +%Y%m%d_%H%M%S).sql.gz
	@echo "$(GREEN)Бэкап создан$(NC)"

		--cov=src \
		--cov-report=html \

	@$(DC) exec redis redis-cli

	@curl -s http://localhost:8000/health | python3 -m json.tool
```
